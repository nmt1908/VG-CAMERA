# ==========================================
# SUPER SCRIPT: SETUP MÁY CHUYÊN DỤNG VG-CAMERA
# Copy nguyên khối script này ném vào Terminal
# ==========================================

# ----- 1. CẤP QUYỀN CAMERA -----
adb shell pm grant com.example.vgcamera android.permission.CAMERA
adb shell appops set com.example.vgcamera CAMERA allow

# ----- 2. CẤP QUYỀN MIC (THU ÂM VIDEO) -----
adb shell pm grant com.example.vgcamera android.permission.RECORD_AUDIO
adb shell appops set com.example.vgcamera RECORD_AUDIO allow

# ----- 3. CẤP QUYỀN BỘ NHỚ LƯU TRỮ (ANDROID 10 TRỞ XUỐNG) -----
adb shell pm grant com.example.vgcamera android.permission.READ_EXTERNAL_STORAGE
adb shell pm grant com.example.vgcamera android.permission.WRITE_EXTERNAL_STORAGE
adb shell appops set com.example.vgcamera READ_EXTERNAL_STORAGE allow
adb shell appops set com.example.vgcamera WRITE_EXTERNAL_STORAGE allow

# ----- 4. CẤP QUYỀN BỘ NHỚ CHI TIẾT (ANDROID 13 TRỞ LÊN) -----
adb shell pm grant com.example.vgcamera android.permission.READ_MEDIA_IMAGES
adb shell pm grant com.example.vgcamera android.permission.READ_MEDIA_VIDEO
adb shell appops set com.example.vgcamera READ_MEDIA_IMAGES allow
adb shell appops set com.example.vgcamera READ_MEDIA_VIDEO allow

# ----- 5. CẤP SIÊU QUYỀN QUẢN LÝ / XÓA RÁC (ANDROID 11 TRỞ LÊN) -----
adb shell appops set com.example.vgcamera MANAGE_EXTERNAL_STORAGE allow

# ----- 6. CẤP QUYỀN VỊ TRÍ CHÍNH XÁC (GPS) -----
adb shell pm grant com.example.vgcamera android.permission.ACCESS_FINE_LOCATION
adb shell pm grant com.example.vgcamera android.permission.ACCESS_COARSE_LOCATION
adb shell appops set com.example.vgcamera FINE_LOCATION allow
adb shell appops set com.example.vgcamera COARSE_LOCATION allow

# ----- 7. CẤP QUYỀN WIFI LÂN CẬN (ANDROID 13+) -----
adb shell pm grant com.example.vgcamera android.permission.NEARBY_WIFI_DEVICES
adb shell appops set com.example.vgcamera NEARBY_WIFI_DEVICES allow
